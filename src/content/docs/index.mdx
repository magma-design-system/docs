<Fragment slot="react">

```tsx
import { MdsButton, MdsInput, MdsInputField } from '@maggioli-design-system/magma-react';

export function Subscribe() {
  return (
    <>
      <MdsInputField label="Email">
        <MdsInput
          name="email"
          type="email"
          placeholder="name@example.com"
          onMdsInputChange={(event) => console.log(event.detail.value)}
        />
      </MdsInputField>
      <MdsButton label="Send" icon="mi/baseline/send" variant="primary" />
    </>
  );
}
```

</Fragment>
<Fragment slot="angular">

```ts
import { Component } from '@angular/core';
import { MdsButton, MdsInput, MdsInputField } from '@maggioli-design-system/magma-angular';

@Component({
  selector: 'app-subscribe',
  imports: [MdsButton, MdsInput, MdsInputField],
  template: `
    <mds-input-field label="Email">
      <mds-input
        name="email"
        type="email"
        placeholder="name@example.com"
        (mdsInputChange)="onChange($event)"
      ></mds-input>
    </mds-input-field>
    <mds-button label="Send" icon="mi/baseline/send" variant="primary"></mds-button>
  `,
})
export class SubscribeComponent {
  onChange(event: CustomEvent) {
    console.log(event.detail.value);
  }
}
```
